(function () {
  var MODEL = /*OMG_MODEL*/null;
  var dirty = !!(MODEL && MODEL.dirty); // true - подставлен шаблон заполнения, ещё не записан
  var controls = []; // {field, get: function () -> value, el}
  var lastText = null; // последнее текстовое поле в фокусе - туда вставляются готовые фразы

  function el(tag, cls, text) {
    var e = document.createElement(tag);
    if (cls) { e.className = cls; }
    if (text !== undefined && text !== null) { e.appendChild(document.createTextNode(text)); }
    return e;
  }
  function on(e, name, fn) {
    if (e.addEventListener) { e.addEventListener(name, fn, false); } else { e.attachEvent('on' + name, fn); }
  }
  function touch() { dirty = true; }
  function autosize(t) {
    t.style.height = 'auto';
    t.style.height = (t.scrollHeight + 2) + 'px';
  }
  function hasClass(e, c) { return (' ' + e.className + ' ').indexOf(' ' + c + ' ') >= 0; }
  function toggleClass(e, c, state) {
    var has = hasClass(e, c);
    if (state === undefined) { state = !has; }
    if (state && !has) { e.className = (e.className + ' ' + c).replace(/^\s+/, ''); }
    if (!state && has) { e.className = (' ' + e.className + ' ').replace(' ' + c + ' ', ' ').replace(/^\s+|\s+$/g, ''); }
  }

  function textControl(f, multi) {
    var box = el('div', 'row');
    var inp;
    if (multi) {
      inp = el('textarea');
      inp.rows = 1;
      on(inp, 'input', function () { autosize(inp); });
    } else {
      inp = el('input');
      inp.type = 'text';
    }
    inp.value = f.value || '';
    on(inp, 'input', touch);
    on(inp, 'change', touch);
    box.appendChild(inp);
    if (f.def) {
      var r = el('button', 'reset', '↺');
      r.type = 'button';
      r.title = 'Вернуть шаблон текста: ' + f.def;
      on(r, 'click', function () { inp.value = f.def; if (multi) { autosize(inp); } touch(); });
      box.appendChild(r);
    }
    if (multi) { setTimeout(function () { autosize(inp); }, 0); }
    return { el: box, get: function () { return inp.value; } };
  }

  function numberControl(f) {
    var box = el('div', 'row');
    var inp = el('input', 'num');
    inp.type = 'text';
    inp.setAttribute('inputmode', 'decimal');
    inp.value = f.value || '';
    on(inp, 'input', touch);
    box.appendChild(inp);
    if (f.unit) { box.appendChild(el('span', 'unit', f.unit)); }
    return { el: box, get: function () { return inp.value.replace(',', '.'); } };
  }

  function dateControl(f) {
    var inp = el('input');
    try { inp.type = 'date'; } catch (e) { inp.type = 'text'; }
    inp.style.width = '180px';
    inp.value = f.value || '';
    on(inp, 'change', touch);
    return { el: inp, get: function () { return inp.value; } };
  }

  function boolControl(f) {
    var lab = el('label');
    var inp = el('input');
    inp.type = 'checkbox';
    inp.checked = (f.value === 'Да' || f.value === 'true');
    on(inp, 'click', touch);
    lab.appendChild(inp);
    lab.appendChild(document.createTextNode(' Да'));
    return { el: lab, get: function () { return inp.checked ? 'Да' : 'Нет'; } };
  }

  function choiceControl(f) {
    var sel = el('select');
    var opts = [''].concat(f.options || []);
    var found = false, i;
    for (i = 0; i < opts.length; i++) {
      var o = el('option', null, opts[i]);
      o.value = opts[i];
      if (opts[i] === f.value) { o.selected = true; found = true; }
      sel.appendChild(o);
    }
    if (!found && f.value) {
      var extra = el('option', null, f.value);
      extra.value = f.value; extra.selected = true;
      sel.appendChild(extra);
    }
    on(sel, 'change', touch);
    return { el: sel, get: function () { return sel.value; } };
  }

  function multiControl(f) {
    var box = el('div', 'chips');
    var current = (f.value || '').split(';');
    var picked = {}, i;
    for (i = 0; i < current.length; i++) { picked[current[i].replace(/^\s+|\s+$/g, '')] = true; }
    var inputs = [];
    for (i = 0; i < (f.options || []).length; i++) {
      (function (opt) {
        var lab = el('label');
        var inp = el('input');
        inp.type = 'checkbox';
        inp.checked = !!picked[opt];
        toggleClass(lab, 'on', inp.checked);
        on(inp, 'click', function () { toggleClass(lab, 'on', inp.checked); touch(); });
        lab.appendChild(inp);
        lab.appendChild(document.createTextNode(opt));
        box.appendChild(lab);
        inputs.push({ inp: inp, opt: opt });
      })(f.options[i]);
    }
    return { el: box, get: function () {
      var res = [], k;
      for (k = 0; k < inputs.length; k++) { if (inputs[k].inp.checked) { res.push(inputs[k].opt); } }
      return res.join('; ');
    } };
  }

  function tableControl(f) {
    var box = el('div');
    var cols = (f.columns && f.columns.length) ? f.columns : ['Значение'];
    var table = el('table', 'grid');
    var thead = el('thead'), tr = el('tr'), i;
    for (i = 0; i < cols.length; i++) { tr.appendChild(el('th', null, cols[i])); }
    tr.appendChild(el('th'));
    thead.appendChild(tr);
    table.appendChild(thead);
    var tbody = el('tbody');
    table.appendChild(tbody);

    function addRow(values) {
      var row = el('tr'), c;
      for (c = 0; c < cols.length; c++) {
        var td = el('td');
        var inp = el('input');
        inp.type = 'text';
        inp.value = (values && values[c]) ? values[c] : '';
        on(inp, 'input', touch);
        td.appendChild(inp);
        row.appendChild(td);
      }
      var act = el('td', 'act');
      var del = el('button', 'del', '×');
      del.type = 'button';
      del.title = 'Удалить строку';
      on(del, 'click', function () { tbody.removeChild(row); touch(); });
      act.appendChild(del);
      row.appendChild(act);
      tbody.appendChild(row);
    }
    var rows = f.rows || [];
    for (i = 0; i < rows.length; i++) { addRow(rows[i]); }
    if (!rows.length) { addRow(null); }

    var add = el('button', 'btn', '+ Добавить строку');
    add.type = 'button';
    on(add, 'click', function () { addRow(null); touch(); });
    box.appendChild(table);
    box.appendChild(add);

    return { el: box, get: function () {
      var res = [], r, c, trs = tbody.getElementsByTagName('tr');
      for (r = 0; r < trs.length; r++) {
        var ins = trs[r].getElementsByTagName('input'), cells = [], any = false;
        for (c = 0; c < ins.length; c++) { cells.push(ins[c].value); if (ins[c].value) { any = true; } }
        if (any) { res.push(cells); }
      }
      return res;
    } };
  }

  // Оформление шаблона (размер в пунктах, как в печатной форме).
  function px(pt) { return Math.round(pt * 4 / 3) + 'px'; }
  var ALIGN = { 'Центр': 'center', 'Право': 'right', 'Ширина': 'justify', 'Лево': 'left' };
  function applyStyle(node, f) {
    if (f.bold) { node.style.fontWeight = '700'; }
    if (f.italic) { node.style.fontStyle = 'italic'; }
    if (f.size) { node.style.fontSize = px(f.size); }
    if (f.align && ALIGN[f.align]) { node.style.textAlign = ALIGN[f.align]; }
  }
  function styleControls(box, f) {
    var tags = ['input', 'textarea', 'select'], i, j, list;
    for (i = 0; i < tags.length; i++) {
      list = box.getElementsByTagName(tags[i]);
      for (j = 0; j < list.length; j++) { applyStyle(list[j], f); }
    }
  }

  function buildField(f) {
    if (f.type === 'Пояснение') {
      var note = el('div', 'field note', f.value);
      applyStyle(note, f);
      return note;
    }
    var row = el('div', 'field');
    var lab = el('label', 'title', f.title);
    if (f.required) { lab.appendChild(el('span', 'req', '*')); }
    row.appendChild(lab);
    var ctlBox = el('div', 'ctl');
    var c;
    switch (f.type) {
      case 'Строка': c = textControl(f, false); break;
      case 'Число': c = numberControl(f); break;
      case 'Дата': c = dateControl(f); break;
      case 'Флажок': c = boolControl(f); break;
      case 'Выбор': c = choiceControl(f); break;
      case 'ВыборНесколько': c = multiControl(f); break;
      case 'Таблица': c = tableControl(f); break;
      default: c = textControl(f, true);
    }
    ctlBox.appendChild(c.el);
    styleControls(ctlBox, f);
    row.appendChild(ctlBox);
    controls.push({ field: f, get: c.get, row: row });
    row.setAttribute('data-param', f.param || '');
    return row;
  }

  // Прокрутка колонки .main к блоку (offsetTop считается от .main, у неё position: relative).
  function scrollToBox(node) {
    var main = document.getElementById('main'), top = 0;
    while (node && node !== main) { top += node.offsetTop; node = node.offsetParent; }
    main.scrollTop = Math.max(0, top - 8);
  }

  function build() {
    var main = document.getElementById('main');
    var nav = document.getElementById('nav');
    if (!MODEL) { main.appendChild(el('div', 'empty', 'Выберите шаблон протокола')); return; }
    if (MODEL.font) { document.body.style.fontFamily = '"' + MODEL.font + '", "Segoe UI", Arial, sans-serif'; }
    if (MODEL.size) { document.body.style.fontSize = px(MODEL.size + 1); }

    var head = el('div', 'head');
    if (MODEL.logo) { var img = el('img'); img.src = MODEL.logo; head.appendChild(img); }
    if (MODEL.org) { head.appendChild(el('div', 'org', MODEL.org)); }
    for (var k = 0; MODEL.orgLines && k < MODEL.orgLines.length; k++) { head.appendChild(el('div', 'orgline', MODEL.orgLines[k])); }
    if (MODEL.title) { head.appendChild(el('h1', null, MODEL.title)); }
    var meta = el('div', 'meta'), i, j;
    for (i = 0; i < MODEL.header.length; i++) {
      if (!MODEL.header[i].value) { continue; }
      var s = el('span', null, MODEL.header[i].label + ': ');
      s.appendChild(el('b', null, MODEL.header[i].value));
      meta.appendChild(s);
    }
    head.appendChild(meta);

    // Боковая панель разделов скрывается кнопкой; выбор запоминается в браузере врача.
    var layout = document.getElementById('layout');
    var toggle = el('button', 'navtoggle');
    toggle.type = 'button';
    function setNav(show) {
      toggleClass(layout, 'nonav', !show);
      toggle.innerHTML = '';
      toggle.appendChild(document.createTextNode(show ? 'Скрыть разделы' : 'Разделы'));
      try { window.localStorage.setItem('omgProtocolNav', show ? '1' : '0'); } catch (x) { }
    }
    var showNav = true;
    try { showNav = window.localStorage.getItem('omgProtocolNav') !== '0'; } catch (x) { }
    on(toggle, 'click', function () { setNav(hasClass(layout, 'nonav')); });
    setNav(showNav);
    head.appendChild(toggle);
    main.appendChild(head);

    for (i = 0; i < MODEL.sections.length; i++) {
      (function (sec, idx) {
        var box = el('div', 'section');
        box.id = 'sec' + idx;
        var h = el('h2', null, sec.title);
        applyStyle(h, { size: sec.size, align: sec.align });

        box.appendChild(h);
        var fields = el('div', 'fields');
        for (j = 0; j < sec.fields.length; j++) { fields.appendChild(buildField(sec.fields[j])); }
        box.appendChild(fields);
        main.appendChild(box);

        var a = el('a', null, sec.title);
        a.href = '#sec' + idx;
        a.appendChild(el('span', 'badge', String(sec.fields.length)));
        on(a, 'click', function (e) {
          if (e && e.preventDefault) { e.preventDefault(); }
          toggleClass(box, 'collapsed', false);
          scrollToBox(box);
          var links = nav.getElementsByTagName('a'), k;
          for (k = 0; k < links.length; k++) { toggleClass(links[k], 'active', links[k] === a); }
        });
        nav.appendChild(a);
      })(MODEL.sections[i], i);
    }
  }

  function jsonString(s) {
    return '"' + String(s).replace(/[\\"\u0000-\u001f\u2028\u2029]/g, function (ch) {
      switch (ch) {
        case '"': return '\\"';
        case '\\': return '\\\\';
        case '\n': return '\\n';
        case '\r': return '\\r';
        case '\t': return '\\t';
        default: return '\\u' + ('0000' + ch.charCodeAt(0).toString(16)).slice(-4);
      }
    }) + '"';
  }
  function toJson(v) {
    if (typeof JSON !== 'undefined' && JSON.stringify) { return JSON.stringify(v); }
    var i, parts = [];
    if (v instanceof Array) {
      for (i = 0; i < v.length; i++) { parts.push(toJson(v[i])); }
      return '[' + parts.join(',') + ']';
    }
    if (typeof v === 'object' && v !== null) {
      for (i in v) { if (v.hasOwnProperty(i)) { parts.push(jsonString(i) + ':' + toJson(v[i])); } }
      return '{' + parts.join(',') + '}';
    }
    return jsonString(v === undefined || v === null ? '' : v);
  }

  // Параметр ПараметрыHTML поля, в котором стоит курсор: по нему 1С отбирает готовые фразы.
  var lastParam = '';
  function fieldParam(t) {
    while (t && t.getAttribute) {
      var p = t.getAttribute('data-param');
      if (p !== null) { return p; }
      t = t.parentNode;
    }
    return null;
  }
  function trackFocus(e) {
    var t = (e && (e.target || e.srcElement)) || null;
    if (t && (t.tagName === 'TEXTAREA' || (t.tagName === 'INPUT' && t.type === 'text'))) { lastText = t; }
    var p = fieldParam(t);
    if (p !== null) { lastParam = p; }
  }
  if (document.addEventListener) { document.addEventListener('focus', trackFocus, true); }
  else if (document.attachEvent) { document.attachEvent('onfocusin', trackFocus); }

  // ---- API для 1С ----
  // Вставка готовой фразы в позицию курсора последнего текстового поля.
  // mode: ',' - через запятую, ';' - с точкой с запятой, '.' - с точкой, иначе через пробел.
  window.omgInsertText = function (text, mode) {
    var t = lastText;
    if (!t || !t.parentNode) { return 'nofield'; }
    var v = t.value || '';
    var s = typeof t.selectionStart === 'number' ? t.selectionStart : v.length;
    var e = typeof t.selectionEnd === 'number' ? t.selectionEnd : s;
    var before = v.substring(0, s);
    var ins = String(text || '').replace(/^\s+|\s+$/g, '');
    if (mode === ',') { ins = (before.replace(/\s+$/, '') ? ', ' : '') + ins; }
    else if (mode === ';') { ins = ' ' + ins + ';'; }
    else if (mode === '.') { ins = ' ' + ins + '.'; }
    else if (before && !/\s$/.test(before)) { ins = ' ' + ins; }
    if (!before) { ins = ins.replace(/^\s+/, ''); }
    t.value = before + ins + v.substring(e);
    try { t.selectionStart = t.selectionEnd = before.length + ins.length; } catch (x) { }
    if (t.tagName === 'TEXTAREA') { autosize(t); }
    touch();
    return 'ok';
  };
  window.omgGetData = function () {
    var data = {}, i;
    for (i = 0; i < controls.length; i++) { data[controls[i].field.key] = controls[i].get(); }
    return toJson(data);
  };
  window.omgMissingRequired = function () {
    var res = [], i, first = null;
    for (i = 0; i < controls.length; i++) {
      var c = controls[i], v = c.get();
      var empty = (v instanceof Array) ? v.length === 0 : !String(v).replace(/^\s+|\s+$/g, '') || (c.field.type === 'Флажок' && v === 'Нет');
      var miss = c.field.required && empty;
      toggleClass(c.row, 'missing', miss);
      if (miss) { res.push(c.field.title); if (!first) { first = c.row; } }
    }
    if (first) { scrollToBox(first); }
    return res.join('\n');
  };
  window.omgCurrentParam = function () { return lastParam; };
  window.omgIsDirty = function () { return dirty ? 'true' : 'false'; };
  window.omgSetSaved = function () { dirty = false; };

  build();
})();
