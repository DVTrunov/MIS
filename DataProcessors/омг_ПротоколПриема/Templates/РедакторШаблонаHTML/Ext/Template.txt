<script>
var EDITOR = /*OMG_EDITOR*/null;
(function () {
  var doc = document.getElementById('doc');
  var dirty = false, uid = 0, range = null, action = '';

  function el(tag, cls, text) {
    var e = document.createElement(tag);
    if (cls) { e.className = cls; }
    if (text !== undefined && text !== null) { e.appendChild(document.createTextNode(text)); }
    return e;
  }
  function on(e, name, fn) { if (e.addEventListener) { e.addEventListener(name, fn, false); } else { e.attachEvent('on' + name, fn); } }
  function hasClass(e, c) { return e && e.nodeType === 1 && (' ' + (e.className || '') + ' ').indexOf(' ' + c + ' ') >= 0; }
  function touch() { dirty = true; }
  function isChip(n) { return hasClass(n, 'omg-f'); }

  // ---- JSON без зависимости от window.JSON (старые движки) ----
  function jsonString(s) {
    return '"' + String(s).replace(/[\\"\u0000-\u001f\u2028\u2029]/g, function (ch) {
      var map = { '"': '\\"', '\\': '\\\\', '\n': '\\n', '\r': '\\r', '\t': '\\t' };
      return map[ch] || ('\\u' + ('0000' + ch.charCodeAt(0).toString(16)).slice(-4));
    }) + '"';
  }
  function toJson(v) {
    var i, parts, k;
    if (v === null || v === undefined) { return 'null'; }
    if (typeof v === 'boolean' || typeof v === 'number') { return String(v); }
    if (typeof v === 'string') { return jsonString(v); }
    if (v instanceof Array) {
      parts = [];
      for (i = 0; i < v.length; i++) { parts.push(toJson(v[i])); }
      return '[' + parts.join(',') + ']';
    }
    parts = [];
    for (k in v) { if (v.hasOwnProperty(k)) { parts.push(jsonString(k) + ':' + toJson(v[k])); } }
    return '{' + parts.join(',') + '}';
  }
  function fromJson(s) {
    if (window.JSON && window.JSON.parse) { return window.JSON.parse(s); }
    return eval('(' + s + ')');
  }
  function newKey() {
    var s = '', i, r;
    for (i = 0; i < 32; i++) {
      r = Math.floor(Math.random() * 16);
      if (i === 12) { r = 4; } else if (i === 16) { r = (r & 3) | 8; }
      s += r.toString(16);
      if (i === 7 || i === 11 || i === 15 || i === 19) { s += '-'; }
    }
    return s;
  }

  // ---- Поля-плашки ----
  function chipData(c) {
    try { return fromJson(c.getAttribute('data-f')); } catch (e) { return { key: c.getAttribute('data-key') || '', name: c.textContent, type: 'Текст' }; }
  }
  function setChip(c, f) {
    if (!f.key) { f.key = newKey(); }
    c.setAttribute('data-key', f.key);
    c.setAttribute('data-f', toJson(f));
    if (!c.getAttribute('data-uid')) { c.setAttribute('data-uid', String(++uid)); }
    c.contentEditable = 'false';
    c.innerHTML = '';
    c.appendChild(document.createTextNode(f.name || 'Поле'));
    var t = 'Поле «' + (f.name || '') + '», ' + (f.typeTitle || f.type || 'Текст');
    if (f.def) { t += '\nПо умолчанию: ' + f.def; }
    if (f.paramName) { t += '\nПараметр: ' + f.paramName; }
    c.title = t + '\nДвойной щелчок - настройки';
  }
  function chips() {
    var list = doc.getElementsByTagName('span'), res = [], i;
    for (i = 0; i < list.length; i++) { if (isChip(list[i])) { res.push(list[i]); } }
    return res;
  }
  // Повтор поля показывается пунктиром; плашки без описания получают его из списка полей.
  function markRefs() {
    var list = chips(), seen = {}, i, f;
    for (i = 0; i < list.length; i++) {
      f = chipData(list[i]);
      var ref = !!(f.key && seen[f.key]);
      if (f.key) { seen[f.key] = true; }
      list[i].className = 'omg-f' + (f.param ? ' param' : '') + (f.type === 'Таблица' ? ' tbl' : '') + (ref ? ' ref' : '');
      if (list[i].contentEditable !== 'false') { list[i].contentEditable = 'false'; }
    }
  }

  // ---- Выделение ----
  function inDoc(node) { while (node) { if (node === doc) { return true; } node = node.parentNode; } return false; }
  function saveRange() {
    var sel = window.getSelection ? window.getSelection() : null;
    if (sel && sel.rangeCount && inDoc(sel.getRangeAt(0).startContainer)) { range = sel.getRangeAt(0).cloneRange(); }
    syncBar();
  }
  function restoreSelection() {
    if (!range) { return; }
    try { doc.focus(); var sel = window.getSelection(); sel.removeAllRanges(); sel.addRange(range); } catch (e) { }
  }
  function upTo(node, tag) {
    while (node && node !== doc) { if (node.nodeType === 1 && node.tagName === tag) { return node; } node = node.parentNode; }
    return null;
  }
  function currentCell() { return range ? (upTo(range.startContainer, 'TD') || upTo(range.startContainer, 'TH')) : null; }

  // ---- Панель ----
  var sizeSel = document.getElementById('size');
  (function () {
    var sizes = ['Размер', 8, 9, 10, 11, 12, 13, 14, 16, 18, 20], i;
    for (i = 0; i < sizes.length; i++) {
      var o = el('option', null, String(sizes[i])); o.value = i ? String(sizes[i]) : ''; sizeSel.appendChild(o);
    }
  })();
  function syncBar() {
    var btns = document.getElementById('bar').getElementsByTagName('button'), i, cmd, st;
    var cell = currentCell();
    for (i = 0; i < btns.length; i++) {
      cmd = btns[i].getAttribute('data-cmd');
      if (btns[i].getAttribute('data-tbl')) { btns[i].disabled = !cell; }
      st = false;
      if (/^(bold|italic|underline|justify)/.test(cmd)) { try { st = document.queryCommandState(cmd); } catch (e) { st = false; } }
      btns[i].className = (btns[i].className.indexOf('main') >= 0 ? 'main' : '') + (st ? ' on' : '');
    }
  }
  // Размер шрифта в пунктах: execCommand даёт font size=7, заменяем его на span со стилем.
  function applySize(pt) {
    restoreSelection();
    try { document.execCommand('styleWithCSS', false, false); } catch (e) { }
    document.execCommand('fontSize', false, '7');
    var fonts = doc.getElementsByTagName('font'), list = [], i;
    for (i = 0; i < fonts.length; i++) { if (fonts[i].getAttribute('size') === '7') { list.push(fonts[i]); } }
    for (i = 0; i < list.length; i++) {
      var s = el('span');
      if (pt) { s.style.fontSize = pt + 'pt'; }
      while (list[i].firstChild) { s.appendChild(list[i].firstChild); }
      list[i].parentNode.replaceChild(s, list[i]);
    }
  }
  function insertNodeAtRange(node) {
    var r = range;
    if (!r || !inDoc(r.startContainer)) { r = document.createRange(); r.selectNodeContents(doc); r.collapse(false); }
    r.deleteContents();
    r.insertNode(node);
    try { r.setStartAfter(node); r.collapse(true); range = r; restoreSelection(); } catch (e) { }
  }
  function tableCommand(cmd) {
    var cell = currentCell();
    if (!cell) { return; }
    var row = cell.parentNode, table = upTo(row, 'TABLE'), idx = cell.cellIndex, i, rows = table.rows;
    if (cmd === 'rowAdd') {
      var nr = row.cloneNode(false);
      for (i = 0; i < row.cells.length; i++) { var c = row.cells[i].cloneNode(false); c.appendChild(el('br')); nr.appendChild(c); }
      row.parentNode.insertBefore(nr, row.nextSibling);
    } else if (cmd === 'colAdd') {
      for (i = 0; i < rows.length; i++) {
        var ref = rows[i].cells[Math.min(idx, rows[i].cells.length - 1)];
        var nc = ref.cloneNode(false); nc.removeAttribute('colspan'); nc.appendChild(el('br'));
        rows[i].insertBefore(nc, ref.nextSibling);
      }
    } else if (cmd === 'rowDel') {
      if (rows.length > 1) { row.parentNode.removeChild(row); } else { table.parentNode.removeChild(table); }
      range = null;
    } else if (cmd === 'colDel') {
      var any = false;
      for (i = 0; i < rows.length; i++) {
        if (rows[i].cells.length > idx) { rows[i].removeChild(rows[i].cells[idx]); }
        if (rows[i].cells.length) { any = true; }
      }
      if (!any) { table.parentNode.removeChild(table); }
      range = null;
    } else if (cmd === 'grid') {
      table.className = hasClass(table, 'grid') ? '' : 'grid';
    }
  }
  function command(cmd) {
    if (cmd === 'field' || cmd === 'param') {
      action = toJson({ act: cmd, text: range ? String(range.toString()) : '' });
      return;
    }
    restoreSelection();
    if (/^(bold|italic|underline|justify)/.test(cmd)) {
      try { document.execCommand('styleWithCSS', false, false); } catch (e) { }
      document.execCommand(cmd, false, null);
    } else if (cmd === 'table') {
      var t = el('table', 'grid'), r, c;
      for (r = 0; r < 2; r++) { var tr = t.insertRow(-1); for (c = 0; c < 2; c++) { tr.insertCell(-1).appendChild(el('br')); } }
      insertNodeAtRange(t);
      var after = el('p'); after.appendChild(el('br'));
      t.parentNode.insertBefore(after, t.nextSibling);
    } else if (cmd === 'hr') {
      insertNodeAtRange(el('hr'));
    } else {
      tableCommand(cmd);
    }
    touch(); saveRange();
  }
  (function () {
    var btns = document.getElementById('bar').getElementsByTagName('button'), i;
    for (i = 0; i < btns.length; i++) {
      (function (btn) {
        // mousedown без фокуса - выделение в тексте не теряется.
        on(btn, 'mousedown', function (e) { if (e.preventDefault) { e.preventDefault(); } });
        on(btn, 'click', function () { command(btn.getAttribute('data-cmd')); });
      })(btns[i]);
    }
    on(sizeSel, 'change', function () { applySize(sizeSel.value); sizeSel.value = ''; touch(); });
  })();

  // ---- События текста ----
  on(doc, 'input', function () { touch(); markRefs(); });
  on(doc, 'keyup', saveRange);
  on(doc, 'mouseup', saveRange);
  on(doc, 'focus', saveRange);
  on(document, 'selectionchange', function () {
    var sel = window.getSelection ? window.getSelection() : null;
    if (sel && sel.rangeCount && inDoc(sel.getRangeAt(0).startContainer)) { saveRange(); }
  });
  on(doc, 'dblclick', function (e) {
    var t = e.target || e.srcElement;
    while (t && t !== doc && !isChip(t)) { t = t.parentNode; }
    if (!t || t === doc) { return; }
    action = toJson({ act: 'edit', uid: t.getAttribute('data-uid'), f: chipData(t) });
  });
  try { document.execCommand('defaultParagraphSeparator', false, 'p'); } catch (x) { }

  // Шаблон пишут администраторы, но скрипты и обработчики в нём не нужны.
  function sanitize(root) {
    var all = root.getElementsByTagName('*'), i, j, kill = [];
    for (i = 0; i < all.length; i++) {
      var tag = all[i].tagName;
      if (tag === 'SCRIPT' || tag === 'IFRAME' || tag === 'OBJECT' || tag === 'EMBED') { kill.push(all[i]); continue; }
      for (j = all[i].attributes.length - 1; j >= 0; j--) {
        if (/^on/i.test(all[i].attributes[j].name)) { all[i].removeAttribute(all[i].attributes[j].name); }
      }
    }
    for (i = 0; i < kill.length; i++) { kill[i].parentNode.removeChild(kill[i]); }
  }

  function build() {
    var sheet = document.getElementById('sheet');
    if (EDITOR && EDITOR.font) { sheet.style.fontFamily = '"' + EDITOR.font + '", Arial, sans-serif'; }
    if (EDITOR && EDITOR.size) { sheet.style.fontSize = EDITOR.size + 'pt'; }
    doc.innerHTML = (EDITOR && EDITOR.html) || '<p><br></p>';
    sanitize(doc);
    var list = chips(), i, fields = {}, given = (EDITOR && EDITOR.fields) || [];
    for (i = 0; i < given.length; i++) { fields[given[i].key] = given[i]; }
    for (i = 0; i < list.length; i++) {
      var key = list[i].getAttribute('data-key') || '';
      var f = fields[key] || { key: key, name: list[i].textContent, type: 'Текст' };
      setChip(list[i], f);
    }
    markRefs();
    syncBar();
  }

  // ---- API для 1С ----
  window.omgEdIsDirty = function () { return dirty ? 'true' : 'false'; };
  window.omgEdSetSaved = function () { dirty = false; };
  // Действие после щелчка (вставить поле / параметр, настроить поле) - 1С забирает его один раз.
  window.omgEdTakeAction = function () { var a = action; action = ''; return a; };
  // HTML шаблона: плашки в виде <span class="omg-f" data-key="...">Имя</span>; описания полей - отдельно.
  window.omgEdGetData = function () {
    var copy = doc.cloneNode(true), list = copy.getElementsByTagName('span'), found = [], i, fields = [], seen = {};
    for (i = 0; i < list.length; i++) { if (isChip(list[i])) { found.push(list[i]); } }
    for (i = 0; i < found.length; i++) {
      var f = chipData(found[i]);
      if (!f.key) { f.key = newKey(); }
      if (!seen[f.key]) { seen[f.key] = true; fields.push(f); }
      found[i].parentNode.replaceChild(document.createTextNode('\u0001' + f.key + '\u0002' + (f.name || '') + '\u0003'), found[i]);
    }
    var html = copy.innerHTML.replace(/\u0001([^\u0002]*)\u0002([^\u0003]*)\u0003/g, function (m, k, n) {
      return '<span class="omg-f" data-key="' + k + '">' + n + '</span>';
    });
    return toJson({ html: html, fields: fields });
  };
  window.omgEdInsertField = function (json) {
    var f = fromJson(json), chip = el('span', 'omg-f');
    setChip(chip, f);
    insertNodeAtRange(chip);
    var space = document.createTextNode(' ');
    insertNodeAtRange(space);
    touch(); markRefs();
    return 'ok';
  };
  // Новые настройки поля - у всех его плашек (и повторов).
  window.omgEdUpdateField = function (uidValue, json) {
    var f = fromJson(json), list = chips(), i, target = null, old;
    for (i = 0; i < list.length; i++) { if (list[i].getAttribute('data-uid') === String(uidValue)) { target = list[i]; } }
    if (!target) { return 'nofield'; }
    old = chipData(target);
    if (!f.key) { f.key = old.key; }
    for (i = 0; i < list.length; i++) {
      if (list[i] === target || (old.key && chipData(list[i]).key === old.key)) { setChip(list[i], f); }
    }
    touch(); markRefs();
    return 'ok';
  };

  build();
})();
</script>
