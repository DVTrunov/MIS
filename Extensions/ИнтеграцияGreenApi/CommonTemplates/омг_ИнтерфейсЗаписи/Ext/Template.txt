        // Функция переключения языков (остается без изменений)
function switchLang(lang) {
    document.getElementById('btn-ru').classList.remove('active');
    document.getElementById('btn-kk').classList.remove('active');
    document.getElementById('btn-' + lang).classList.add('active');

    document.getElementById('content-ru').classList.remove('active-lang');
    document.getElementById('content-kk').classList.remove('active-lang');
    document.getElementById('content-' + lang).classList.add('active-lang');
}

// Умная функция добавления в календарь
function addToCalendar(title, dateStr, timeStr, doctor, address) {
    const dateParts = dateStr.split('.');
    const timeParts = timeStr.split(':');
    
    // Формируем время начала
    const startStr = `${dateParts[2]}${dateParts[1]}${dateParts[0]}T${timeParts[0]}${timeParts[1]}00`;
    
    // Прибавляем 30 минут для времени окончания
    let endHour = parseInt(timeParts[0]);
    let endMinute = parseInt(timeParts[1]) + 30;
    if (endMinute >= 60) { endMinute -= 60; endHour += 1; }
    const endStr = `${dateParts[2]}${dateParts[1]}${dateParts[0]}T${String(endHour).padStart(2, '0')}${String(endMinute).padStart(2, '0')}00`;

    // Определяем, iPhone/iPad ли это
    const isIOS = /iPad|iPhone|iPod/.test(navigator.userAgent) || (navigator.platform === 'MacIntel' && navigator.maxTouchPoints > 1);

    if (isIOS) {
        // Для iOS отдаем .ics (Safari откроет нативное окно добавления события)
        const icsData = `BEGIN:VCALENDAR\nVERSION:2.0\nPRODID:-//1C Clinic//RU\nBEGIN:VEVENT\nSUMMARY:${title}\nDESCRIPTION:Врач / Дәрігер: ${doctor}\nLOCATION:${address}\nDTSTART:${startStr}\nDTEND:${endStr}\nEND:VEVENT\nEND:VCALENDAR`;

        const blob = new Blob([icsData], { type: 'text/calendar;charset=utf-8' });
        const link = document.createElement('a');
        link.href = window.URL.createObjectURL(blob);
        link.setAttribute('download', 'appointment.ics');
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
    } else {
        // Для Android и ПК открываем ссылку Google Календаря
        const details = encodeURIComponent("Врач / Дәрігер: " + doctor);
        const loc = encodeURIComponent(address);
        const text = encodeURIComponent(title);
        
        const googleCalUrl = `https://calendar.google.com/calendar/render?action=TEMPLATE&text=${text}&dates=${startStr}/${endStr}&details=${details}&location=${loc}`;
        
        window.open(googleCalUrl, '_blank');
    }
}
