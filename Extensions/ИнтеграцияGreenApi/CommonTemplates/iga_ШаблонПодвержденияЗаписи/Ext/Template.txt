    <script>
        // Функция переключения языков остается без изменений
		function switchLang(lang) {
		    document.getElementById('btn-ru').classList.remove('active');
		    document.getElementById('btn-kk').classList.remove('active');
		    document.getElementById('btn-' + lang).classList.add('active');
		    document.getElementById('content-ru').classList.remove('active-lang');
		    document.getElementById('content-kk').classList.remove('active-lang');
		    document.getElementById('content-' + lang).classList.add('active-lang');
		}

		// Глобальная переменная для хранения данных клика
		let pendingAction = null;

		// 1. Функция, которая вызывается по клику на кнопку в карточке
		function sendDecision(btnElement, method, id, num, promptText) {
		    // Сохраняем все данные, чтобы использовать их после подтверждения
		    pendingAction = { btnElement, method, id, num };
		    
		    // Подставляем текст из 1С в модальное окно
		    document.getElementById('modal-text').innerText = promptText;
		    
		    // Показываем окно с анимацией
		    let modal = document.getElementById('custom-modal');
		    modal.style.display = 'flex';
		    // Небольшая задержка для срабатывания CSS-анимации
		    setTimeout(() => modal.classList.add('show'), 10);
		}

		// 2. Функция отмены (скрывает окно)
		function closeModal() {
		    let modal = document.getElementById('custom-modal');
		    modal.classList.remove('show');
		    setTimeout(() => { 
		        modal.style.display = 'none'; 
		        pendingAction = null; 
		    }, 300); // 300мс - время анимации
		}

		// 3. Функция, которая делает реальный запрос в 1С (вызывается из модального окна)
		async function executeDecision() {
		    // Скрываем окно
		    closeModal();
		    
		    // Если данных нет, выходим
		    if (!pendingAction) return;

		    let { btnElement, method, id, num } = pendingAction;

		    let actionsRu = document.getElementById('actions-ru-' + num);
		    let actionsKk = document.getElementById('actions-kk-' + num);
		    
		    if (actionsRu) actionsRu.querySelectorAll('.btn-action').forEach(b => b.disabled = true);
		    if (actionsKk) actionsKk.querySelectorAll('.btn-action').forEach(b => b.disabled = true);
		    
		    let originalText = btnElement.innerText;
		    btnElement.innerText = "⏳...";

		    try {
		        // !!! НЕ ЗАБУДЬТЕ УКАЗАТЬ ИМЯ ВАШЕЙ БАЗЫ В URL !!!
		        let response = await fetch('/ИмяВашейБазы/hs/api/appointments', {
		            method: 'POST',
		            headers: { 'Content-Type': 'application/json' },
		            body: JSON.stringify({ metod: method, id: id, num: num })
		        });

		        if (response.ok) {
		            if (actionsRu) actionsRu.style.display = 'none';
		            if (actionsKk) actionsKk.style.display = 'none';
		            
		            if (method === 'confirm') {
		                document.getElementById('status-success-ru-' + num).style.display = 'flex';
		                document.getElementById('status-success-kk-' + num).style.display = 'flex';
		            } else if (method === 'cancel') {
		                document.getElementById('status-cancel-ru-' + num).style.display = 'flex';
		                document.getElementById('status-cancel-kk-' + num).style.display = 'flex';
		            } else if (method === 'reschedule') {
		                document.getElementById('status-reschedule-ru-' + num).style.display = 'flex';
		                document.getElementById('status-reschedule-kk-' + num).style.display = 'flex';
		            }
		        } else {
		            alert("Ошибка на сервере"); // Это оставим, системный алерт для ошибки не так страшен
		            btnElement.innerText = originalText;
		            if (actionsRu) actionsRu.querySelectorAll('.btn-action').forEach(b => b.disabled = false);
		            if (actionsKk) actionsKk.querySelectorAll('.btn-action').forEach(b => b.disabled = false);
		        }
		    } catch (error) {
		        alert("Ошибка соединения. Проверьте интернет.");
		        btnElement.innerText = originalText;
		        if (actionsRu) actionsRu.querySelectorAll('.btn-action').forEach(b => b.disabled = false);
		        if (actionsKk) actionsKk.querySelectorAll('.btn-action').forEach(b => b.disabled = false);
		    }
		}
    </script>
    <!-- Кастомное модальное окно -->
