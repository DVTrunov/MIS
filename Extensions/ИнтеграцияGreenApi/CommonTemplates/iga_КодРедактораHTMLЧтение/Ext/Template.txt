		    <script>
		        var viewer = document.getElementById('viewer');
		
		        // Единственная необходимая функция - прием текста из 1С
        function setText(text) {
            if (!text) {
                viewer.innerHTML = ''; 
                return; 
            } 
            
            // Заменяем переносы строк 1С на HTML-переносы 
            var formattedText = text.replace(/(?:\r\n|\r|\n)/g, '<br>'); 
            
            // --- НОВАЯ ЛОГИКА: Форматирование жирного текста (WhatsApp style) ---
            // Регулярное выражение /\*(.*?)\*/g ищет текст между звездочками
            // $1 подставляет найденный текст внутрь тега <strong>
            formattedText = formattedText.replace(/\*(.*?)\*/g, '<strong>$1</strong>');
            // ------------------------------------------------------------------
            
            viewer.innerHTML = formattedText;
            
            // Натравливаем парсер, чтобы он нашел юникод-смайлы и превратил их в картинки 
            twemoji.parse(viewer); 
        } 
		    </script>
