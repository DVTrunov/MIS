    <script>
        // Определяем правила синтаксиса для встроенного языка 1С (BSL)
        CodeMirror.defineSimpleMode('bsl', {
            start: [
                // 1. Комментарии (начинаются с //)
                {regex: /\/\/.*/, token: 'comment'},
                
                // 2. Строки (в двойных кавычках или начинающиеся с |)
                {regex: /".*?(?:"|$)/, token: 'string'}, 
                {regex: /\|.*/, token: 'string'},
                
                // 3. Открывающие блок (сдвигают следующий код вправо)
                {regex: /(?:Процедура|Функция|Если|Пока|Для|Попытка)(?![а-яА-Яa-zA-Z0-9_])/i, token: 'keyword', indent: true},
                
                // 4. Закрывающие блок (мгновенно сдвигаются влево сами и сдвигают код дальше)
                {regex: /(?:КонецПроцедуры|КонецФункции|КонецЕсли|КонецЦикла|КонецПопытки)(?![а-яА-Яa-zA-Z0-9_])/i, token: 'keyword', dedent: true, dedentIfLineStart: true},
                
                // 5. Промежуточные (сдвигаются влево, а следующую строку сдвигают вправо)
                {regex: /(?:Иначе|ИначеЕсли|Исключение)(?![а-яА-Яa-zA-Z0-9_])/i, token: 'keyword', dedent: true, indent: true, dedentIfLineStart: true},
                
                // 6. Остальные ключевые слова
                {regex: /(?:Тогда|Каждого|Из|Цикл|Перем|Экспорт|Знач|Возврат|Новый|Выполнить|И|Или|Не|То|Далее|Перейти)(?![а-яА-Яa-zA-Z0-9_])/i, token: 'keyword'},
                
                // 7. Булевы значения и Null
                {regex: /(?:Истина|Ложь|Неопределено|NULL)(?![а-яА-Яa-zA-Z0-9_])/i, token: 'keyword'},
                
                // 8. Встроенные функции
                {regex: /(?:Сообщить|СтрДлина|Лев|Прав|Сред|ТекущаяДата|Формат|ЗначениеЗаполнено|ТипЗнч|Тип)(?![а-яА-Яa-zA-Z0-9_])/i, token: 'builtin'},
                
                // 9. Числа
                {regex: /[-+]?\d+(?:\.\d+)?/i, token: 'number'},
                
                // 10. Операторы и пунктуация
                {regex: /[-+\/*=<>()[\]{};:.,]/, token: 'operator'},
                
                // 11. Переменные
                {regex: /[а-яА-Яa-zA-Z_][а-яА-Яa-zA-Z0-9_]*/, token: 'variable'}
            ],
            // СЛУЖЕБНЫЙ БЛОК ДЛЯ УМНОГО ПОВЕДЕНИЯ РЕДАКТОРА
            meta: {
                // Горячая клавиша Ctrl + / для комментирования
                lineComment: "//",
                // "Электрические" слова, при вводе которых строка сразу прыгает на нужный отступ
                electricInput: /^\s*(?:КонецПроцедуры|КонецФункции|КонецЕсли|КонецЦикла|КонецПопытки|Иначе|ИначеЕсли|Исключение)/i
            }
        });

        // Инициализация редактора
        var myEditor = CodeMirror.fromTextArea(document.getElementById('codeArea'), {
            lineNumbers: true,      // Нумерация строк
            matchBrackets: true,    // Подсветка парных скобок
            mode: 'bsl',            // Наш самописный режим для 1С
            theme: 'default',       // Базовая светлая тема
            
            // НАСТРОЙКИ ОТСТУПОВ
            indentUnit: 4,          // Шаг отступа (для расчета ширины)
            tabSize: 4,             // Визуальная ширина одного символа табуляции
            indentWithTabs: true,   // ВАЖНО: Использовать реальный символ Tab (\t) вместо пробелов
            smartIndent: true       // Умные отступы на основе нашего парсера
        });
        
        // Функции для взаимодействия с 1С
        function set1CCode(textCode) { 
            myEditor.setValue(textCode); 
        }
        
        function get1CCode() { 
            return myEditor.getValue(); 
        }
        // Вставляет текст по текущему указателю (или заменяет выделенный текст)
        function insert1CCodeAtCursor(textToInsert) {
            if (myEditor) {
                myEditor.replaceSelection(textToInsert);
                // Обязательно возвращаем фокус в редактор, чтобы можно было сразу продолжить печатать
                myEditor.focus();
            }
        }
    </script>
