      function scrollToBottom() {
        var anchor = document.getElementById('bottom-anchor');
        if (anchor) {
          anchor.scrollIntoView({
            behavior: 'auto',
            block: 'end'
          });
        }
      }

      function clearMessages() {
        var container = document.querySelector('.chat-container');
        if (!container) return;
        
        // Исправлено: удаляем только сами сообщения, оставляя якоря и кнопку загрузки истории
        var messages = container.querySelectorAll('.message');
        messages.forEach(function(msg) {
            msg.remove();
        });

        if (typeof clearQuote === "function") {
          clearQuote();
        }
        console.log("Чат очищен");
      }

      window.onload = function() {
        setTimeout(scrollToBottom, 100);
      };

		function appendMessage(jsonMessage, prepend) {
		  var data = JSON.parse(jsonMessage);
		  var container = document.querySelector('.chat-container');
		  var anchor = document.getElementById('bottom-anchor');
		  var loadingBlock = document.getElementById('history-loading-block');
		  if (!container || !anchor) return;

		  // Кроссбраузерный доступ к корневому элементу, который реально скроллится
		  var scrollElement = document.documentElement || document.body;
		  var oldScrollHeight = 0;

		  // --- 1. ФИКСАЦИЯ ВЫСОТЫ ДО ВСТАВКИ ---
		  if (prepend === true) {
		    oldScrollHeight = scrollElement.scrollHeight;
		  }

		  // --- 2. СОЗДАНИЕ HTML ---
		  var msgDiv = document.createElement('div');
		  msgDiv.id = 'msg_' + data.id;
		  msgDiv.className = 'message ' + (data.isIncoming ? 'incoming' : 'outgoing');
		  
		  var html = '';
		  if (data.isSystem) {
		    msgDiv.className = 'system-message';
		    html = data.text;
		  } else {
		    var cleanText = data.text ? data.text.replace(/'/g, "\\'").replace(/"/g, '&quot;') : "";
		    var authorName = data.isIncoming ? 'Клиент' : 'Вы';
		    
		    html += '<div class="reply-btn" title="Ответить" onclick="setReply(\'' + data.id + '\', \'' + cleanText + '\', \'' + authorName + '\')">↩</div>';
		    
		    if (data.quoteText) {
		      var displayQuote = data.quoteText;
		      if (typeof twemoji !== 'undefined') {
		        displayQuote = twemoji.parse(data.quoteText, { folder: '72x72', ext: '.png' });
		      }
		      html += '<div class="quoted-block">' +
		                '<span class="quoted-author">' + (data.quoteAuthor ? data.quoteAuthor : '') + '</span>' +
		                '<span class="quoted-text">' + displayQuote + '</span>' +
		              '</div>';
		    }
		    
	        // 3. Подпись оператора [cite: 304]
	        if (!data.isIncoming && data.operatorName) {
	            html += '<span class="message-operator">Вы (' + data.operatorName + '):</span>';
	        }

		    if (data.hasFile || data.type === 'image') {
		      var iconClass = data.iconClass || 'icon-save';
		      html += '<div class="message-media">';
		      if (data.url) {
		        html += '<img class="message-image" src="' + data.url + '" onclick="imageClicked(\'' + data.id + '\')" />';
		      } else if (data.thumbnail) {
		        html += '<img src="data:image/jpeg;base64,' + data.thumbnail + '" class="media-thumb" />';
		      }
		      html += '<button class="btn-media ' + iconClass + '" onclick="viewMedia(\'' + data.id + '\')"></button>';
		      html += '</div>';
		    }
		    
		    if (data.text) {
		      var displayText = data.text;
		      if (typeof twemoji !== 'undefined') {
		        displayText = twemoji.parse(data.text, { folder: '72x72', ext: '.png' });
		      }
		      html += '<div class="message-text">' + displayText + '</div>';
		    }
		    html += '<span class="message-time">' + data.time + '</span>';
		  }
		  msgDiv.innerHTML = html;
		  
		  // --- 3. ВСТАВКА В ДОКУМЕНТ И СИНХРОННЫЙ СДВИГ ---
		  if (prepend === true) {
		    // Вставляем новое сообщение
		    if (loadingBlock && loadingBlock.nextSibling) {
		      container.insertBefore(msgDiv, loadingBlock.nextSibling);
		    } else {
		      container.insertBefore(msgDiv, container.firstChild);
		    }
		    
		    // Принудительно заставляем браузер пересчитать высоту документа прямо сейчас
		    var newScrollHeight = scrollElement.scrollHeight;
		    var delta = newScrollHeight - oldScrollHeight;
		    
		    // Если страница реально выросла, крутим ползунок вниз ровно на эту разницу
		    if (delta > 0) {
		      window.scrollBy(0, delta);
		    }
		    
		  } else {
		    // Добавление новых сообщений вниз
		    container.insertBefore(msgDiv, anchor);
		    if (!data.isHistory) {
		      setTimeout(scrollToBottom, 50);
		    }
		  }
		}

      function scrollEnd() {
        setTimeout(scrollToBottom, 50);
      }

      function setReply(messageId, messageText, author) {
        document.getElementById('quote-preview-author').innerText = author;
        document.getElementById('quote-preview-text').innerText = messageText;
        document.getElementById('quote-preview').style.display = 'flex';
        document.querySelector('.chat-container').style.paddingBottom = "60px";
        scrollToBottom();
        document.title = "v8:setQuoteId:" + messageId;
      }

      function cancelQuote() {
        clearQuote();
        document.title = "v8:cancelQuote";
      }

      function clearQuote() {
        document.getElementById('quote-preview').style.display = 'none';
        document.querySelector('.chat-container').style.paddingBottom = "0px";
      }

      function viewMedia(messageId) {
        document.title = "v8:viewMedia:" + messageId;
      }

      function requestMoreHistory() {
        document.title = "v8:loadMoreHistory:" + Date.now();
      }

      function setHistoryButtonVisibility(visible) {
        var block = document.getElementById('history-loading-block');
        if (block) {
          if (visible) {
            block.classList.remove('hidden');
          } else {
            block.classList.add('hidden');
          }
        }
      }
